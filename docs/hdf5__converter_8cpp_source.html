48 void hdf5_write_signal_header(const H5::H5File &file, const SignalHeader &header) {
49 H5::Group data_group = file.createGroup("/data");
50 data_group.createAttribute("format_name", H5::StrType(0, 6), H5::DataSpace(H5S_SCALAR))
51 .write(H5::StrType(0, 6), "BINHDR");
52 data_group
53 .createAttribute("format_version", H5::StrType(0, H5T_VARIABLE),
54 H5::DataSpace(H5S_SCALAR))
55 .write(H5::StrType(0, H5T_VARIABLE), generate_format_version());
56
57 data_group
58 .createAttribute("frequency", H5::PredType::NATIVE_DOUBLE, H5::DataSpace(H5S_SCALAR))
59 .write(H5::PredType::NATIVE_DOUBLE, &header.frequency);
60 data_group
61 .createAttribute("lsb_default", H5::PredType::NATIVE_DOUBLE, H5::DataSpace(H5S_SCALAR))
62 .write(H5::PredType::NATIVE_DOUBLE, &header.lsb_default);
63 data_group.createAttribute("offset", H5::PredType::NATIVE_INT64, H5::DataSpace(H5S_SCALAR))
64 .write(H5::PredType::NATIVE_INT64, &header.ifirst_point);
65
66 const std::string timestart = header.time_start.empty() ? "" : header.time_start;
67 data_group.createAttribute("time_start", H5::StrType(0, 19), H5::DataSpace(H5S_SCALAR))
68 .write(H5::StrType(0, 19), timestart);
69
70 hsize_t dims[] = { static_cast<hsize_t>(header.num_channels) };
71 H5::DataSpace universal_space(1, dims);
72
73 std::vector<const char *> channel_name_ptrs;
74 channel_name_ptrs.reserve(static_cast<size_t>(header.num_channels));
75
76 for (int32_t ch = 0; ch < header.num_channels; ch++) {
77 const auto ch_idx = static_cast<size_t>(ch);
78
79 if (ch_idx < header.channel_names.size() && !header.channel_names[ch_idx].empty()) {
80 channel_name_ptrs.push_back(header.channel_names[ch_idx].c_str());
81 } else {
82 channel_name_ptrs.push_back("");
83 }
84 }
85
86 data_group.createAttribute("channel_names", H5::StrType(0, H5T_VARIABLE), universal_space)
87 .write(H5::StrType(0, H5T_VARIABLE),
88 static_cast<const void *>(channel_name_ptrs.data()));
89
90 if (header.optional_fields.empty()) {
91 return;
92 }
93 for (const auto &[key, value] : header.optional_fields) {
94 if (std::holds_alternative<std::string>(value)) {
95 const auto &str_value = std::get<std::string>(value);
96 data_group
97 .createAttribute(key, H5::StrType(0, H5T_VARIABLE), H5::DataSpace(H5S_SCALAR))
98 .write(H5::StrType(0, H5T_VARIABLE), str_value);
99 } else if (std::holds_alternative<int64_t>(value)) {
100 int64_t int_value = std::get<int64_t>(value);
101 data_group
102 .createAttribute(key, H5::PredType::NATIVE_INT64, H5::DataSpace(H5S_SCALAR))
103 .write(H5::PredType::NATIVE_INT64, &int_value);
104 } else if (std::holds_alternative<double>(value)) {
105 double double_value = std::get<double>(value);
106 data_group
107 .createAttribute(key, H5::PredType::NATIVE_DOUBLE, H5::DataSpace(H5S_SCALAR))
108 .write(H5::PredType::NATIVE_DOUBLE, &double_value);
109 }
110 }

113 void hdf5_write_bin_data(const H5::H5File &file, BinReader &reader, int64_t point_off,
114 int64_t point_cnt) {
115 const int32_t num_channels = reader.header.num_channels;
116 const int64_t points_to_read = normalize_range(reader.total_points, point_off, point_cnt);
117 if (points_to_read == 0) {
118 return;
119 }
120
121 const int64_t block_points = std::min<int64_t>(HDF5_CHUNK_POINTS, points_to_read);
122 hsize_t total_dims[1] = { static_cast<hsize_t>(points_to_read) };
123 hsize_t chunk_dims[1] = { static_cast<hsize_t>(block_points) };
124
125 H5::DSetCreatPropList proplist_params;
126 proplist_params.setChunk(1, chunk_dims);
127 proplist_params.setShuffle();
128 if (H5Zfilter_avail(H5Z_FILTER_DEFLATE) > 0) {
129 proplist_params.setDeflate(HDF5_DEFLATE_LEVEL);
130 }
131
132 const int64_t first_point = reader.header.ifirst_point + point_off;
133
134 std::vector<H5::DataSet> channel_datasets;
135 channel_datasets.reserve(static_cast<size_t>(num_channels));
136
137 H5::Group data_group = file.openGroup("/data");
138
139 for (int32_t ch = 0; ch < num_channels; ch++) {
140 H5::DataSpace dataset_space(1, total_dims);
141 H5::DataSet dataset = data_group.createDataSet(reader.header.channel_names[ch],
142 H5::PredType::NATIVE_INT32,
143 dataset_space, proplist_params);
144
145 const double lsb = (static_cast<size_t>(ch) < reader.header.lsbs.size())
146 ? reader.header.lsbs[static_cast<size_t>(ch)]
147 : reader.header.lsb_default;
148 const std::string unit = static_cast<size_t>(ch) < reader.header.units.size()
149 ? reader.header.units[static_cast<size_t>(ch)]
150 : "";
151
152 dataset.createAttribute("lsb", H5::PredType::NATIVE_DOUBLE, H5S_SCALAR)
153 .write(H5::PredType::NATIVE_DOUBLE, &lsb);
154 dataset.createAttribute("offset", H5::PredType::NATIVE_INT64, H5S_SCALAR)
155 .write(H5::PredType::NATIVE_INT64, &first_point);
156 dataset.createAttribute("type", H5::StrType(0, 5), H5::DataSpace(H5S_SCALAR))
157 .write(H5::StrType(0, 5), "int32");
158 dataset.createAttribute("unit", H5::StrType(0, H5T_VARIABLE), H5S_SCALAR)
159 .write(H5::StrType(0, H5T_VARIABLE), unit);
160
161 channel_datasets.push_back(dataset);
162 }
163
164 std::vector<int32_t> chunk_buffer;
165 std::vector<int32_t> channel_buffer(static_cast<size_t>(block_points));
166 bin_seek(reader, point_off);
167
168 int64_t points_written = 0;
169 while (points_written < points_to_read) {
170 const int64_t request_points =
171 std::min<int64_t>(block_points, points_to_read - points_written);
172 const int64_t points_in_block =
173 bin_read_data_chunk(reader, request_points, chunk_buffer);
174 if (points_in_block == 0)
175 break;
176
177 hsize_t start[1] = { static_cast<hsize_t>(points_written) };
178 hsize_t count[1] = { static_cast<hsize_t>(points_in_block) };
179 H5::DataSpace mem_space(1, count);
180
181 for (int32_t ch = 0; ch < num_channels; ch++) {
182 for (int64_t point = 0; point < points_in_block; point++) {
183 channel_buffer[static_cast<size_t>(point)] =
184 chunk_buffer[(static_cast<size_t>(point) *
185 static_cast<size_t>(num_channels)) +
186 static_cast<size_t>(ch)];
187 }
188
189 H5::DataSpace channel_space = channel_datasets[static_cast<size_t>(ch)].getSpace();
190 channel_space.selectHyperslab(H5S_SELECT_SET, count, start);
191 channel_datasets[static_cast<size_t>(ch)].write(
192 channel_buffer.data(), H5::PredType::NATIVE_INT32, mem_space, channel_space);
193 }
194
195 points_written += points_in_block;
196 }
197
198 if (points_written != points_to_read)
199 throw std::runtime_error("Не удалось записать весь запрошенный диапазон точек в HDF5");
200 }

254 void hdf5_read_signal_header_optional_params(const H5::H5File &file, SignalHeader &metadata) {
255 H5::Group data_group = file.openGroup("/data");
256 for (int i = 0; i < data_group.getNumAttrs(); i++) {
257 H5::Attribute attr = data_group.openAttribute(i);
258 std::string key = attr.getName();
259 H5::DataType type = attr.getDataType();
260
261 if (is_field_optional(attr.getName()))
262 continue;
263
264 if (type.getClass() == H5T_STRING) {
265 std::string str_value;
266 attr.read(type, str_value);
267 metadata.optional_fields[key] = str_value;
268 } else if (type.getClass() == H5T_INTEGER) {
269 int64_t int_value;
270 attr.read(type, &int_value);
271 metadata.optional_fields[key] = int_value;
272 } else if (type.getClass() == H5T_FLOAT) {
273 double double_value;
274 attr.read(type, &double_value);
275 metadata.optional_fields[key] = double_value;
276 }
277 }
278 data_group.close();
279 }

337 int64_t point_off, int64_t point_cnt,
338 int64_t chunk_points) {

341 }

359 std::vector<H5::DataSet> channel_datasets;

364
365 H5::DataSet dataset = data_group.openDataSet(channel_name);
366 channel_datasets.push_back(dataset);
367 }

372 int64_t points_read = 0;

374 const int64_t points_in_block =
375 std::min<int64_t>(actual_chunk, points_to_read - points_read);
376 hsize_t start[1] = { static_cast<hsize_t>(point_off + points_read) };
377 hsize_t count[1] = { static_cast<hsize_t>(points_in_block) };
378
379 H5::DataSpace mem_space(1, count);
380
381 for (int32_t ch = 0; ch < data.num_channels; ch++) {

387 int32_t *dest = data.raw_channel(ch) + points_read;
388 for (int64_t point = 0; point < points_in_block; point++) {
389 dest[point] = channel_buffer[static_cast<size_t>(point)];
390 }
391 }
392
393 points_read += points_in_block;
394 }
