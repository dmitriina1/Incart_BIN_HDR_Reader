67 int64_t point_off, int64_t point_cnt) {
68 if (point_off < 0) {
69 throw std::invalid_argument("Смещение точек не может быть отрицательным");
70 }
71
72 SignalHeader metadata;
73 duckdb_read_signal_metadata(conn, metadata);
74
75 const int64_t total_points = std::max<int64_t>(0, metadata.total_points());
76 const int64_t points_to_read =
77 prepare_signal_data_window(metadata, total_points, point_off, point_cnt, data, false);
78
79 if (points_to_read == 0) {
80 return;
81 }
82
83 std::string sql_query = create_select_query_for_signal_data(data);
84
85 duckdb_prepared_statement stmt;
86 if (duckdb_prepare(conn, sql_query.c_str(), &stmt) == DuckDBError) {
87 const char *prepare_error = duckdb_prepare_error(stmt);
88 std::string error_message = (prepare_error && prepare_error[0] != '\0')
89 ? prepare_error
90 : "DuckDB не вернул текст ошибки";
91 duckdb_destroy_prepare(&stmt);
92 throw std::runtime_error(
93 std::format("Не удалось подготовить запрос для чтения точек: {}", error_message));
94 }
95
96 duckdb_bind_int64(stmt, 1, data.ifirst_point);
97 duckdb_bind_int64(stmt, 2, data.ilast_point);
98
99 duckdb_result res;
100
101 if (duckdb_execute_prepared_streaming(stmt, &res) == DuckDBError) {
102 std::string error_message =
103 duckdb_result_error_message(res, "DuckDB не вернул текст ошибки");
104 duckdb_destroy_result(&res);
105 duckdb_destroy_prepare(&stmt);
106 throw std::runtime_error(
107 std::format("Не удалось выполнить запрос для чтения точек: {}", error_message));
108 }
109
110 duckdb_data_chunk chunk;
111 int64_t offset = 0;
112
113 transfer_chunk_to_channel_data(chunk, res, data, stmt, offset, points_to_read);
114
115 duckdb_destroy_result(&res);
116 duckdb_destroy_prepare(&stmt);
117
118 if (offset != points_to_read) {
119 throw std::runtime_error(
120 std::format("Не удалось прочитать весь диапазон, ожидалось {} точек, а получили {}",
121 points_to_read, offset));

153 static void duckdb_write_signal_data_chunked(const std::filesystem::path &bin_path,
154 const std::filesystem::path &duckdb_path,
155 int64_t point_off, int64_t point_cnt,
156 int64_t chunk_points,
157 int64_t commit_every_n_points) {
158 BinReader reader;
159 bin_open(bin_path, reader);
160
161 duckdb_database db = nullptr;
162 duckdb_connection conn = nullptr;
163 duckdb_config config = nullptr;
164
165 open_and_set_config(config, duckdb_path, db, reader, conn);
166
167 try {
168 if (chunk_points <= 0)
169 throw std::invalid_argument("Количество точек в чанке должно быть больше нуля");
170 if (point_off < 0)
171 throw std::invalid_argument("Смещение точек должно быть неотрицательным");
172 if (reader.header.num_channels <= 0) {
173 bin_close(reader);
174 throw std::runtime_error(
175 "Количество каналов в исходных данных должно быть больше нуля");
176 }
177
178 SignalHeader range_header;
179 const int64_t points_to_read = make_header_for_range(
180 reader.header, reader.total_points, point_off, point_cnt, range_header);
181 const int64_t first_point_id = range_header.ifirst_point;
182
183 create_db_schema(conn, reader.header);
184 duckdb_append_signal_header(conn, range_header);
185 duckdb_append_channels_data(conn, range_header);
186
187 const int64_t block_points = std::min<int64_t>(chunk_points, points_to_read);
188 std::vector<int32_t> chunk_buffer;
189
190 bin_seek(reader, point_off);
191
192 int64_t points_read_total = 0;
193 int64_t points_since_commit = 0;
194 duckdb_appender points_appender = nullptr;
195
196 auto start_transaction = [&]() {
197 duckdb_result res;
198 if (duckdb_query(conn, "BEGIN TRANSACTION;", &res) != DuckDBSuccess) {
199 std::string error_message =
200 duckdb_result_error_message(res, "DuckDB не вернул текст ошибки");
201 duckdb_destroy_result(&res);
202 throw std::runtime_error(
203 std::format("Не удалось начать транзакцию: {}", error_message));
204 }
205 duckdb_destroy_result(&res);
206
207 if (duckdb_appender_create(conn, nullptr, "points", &points_appender) ==
208 DuckDBError)
209 throw std::runtime_error(
210 "Не удалось создать DuckDB appender для таблицы points");
211 };
212
213 auto commit_and_reopen = [&]() {
214 duckdb_result res;
215 if (duckdb_appender_flush(points_appender) == DuckDBError) {
216 duckdb_error_data error = duckdb_appender_error_data(points_appender);
217 std::string error_message = duckdb_error_data_message(error);
218 duckdb_destroy_error_data(&error);
219 throw std::runtime_error(std::format(
220 "Не удалось сбросить DuckDB appender points: {}", error_message));
221 }
222 duckdb_appender_destroy(&points_appender);
223
224 if (duckdb_query(conn, "COMMIT;", &res) != DuckDBSuccess) {
225 std::string error_message =
226 duckdb_result_error_message(res, "DuckDB не вернул текст ошибки");
227 duckdb_destroy_result(&res);
228 throw std::runtime_error(
229 std::format("Не удалось зафиксировать транзакцию: {}", error_message));
230 }
231 duckdb_destroy_result(&res);
232 };
233
234 start_transaction();
235
236 while (points_read_total < points_to_read) {
237 const int64_t request_points =
238 std::min<int64_t>(block_points, points_to_read - points_read_total);
239 const int64_t points_in_block =
240 bin_read_data_chunk(reader, request_points, chunk_buffer);
241 if (points_in_block == 0)
242 break;
243
244 append_points_block(points_appender, chunk_buffer, points_in_block,
245 reader.header.num_channels, first_point_id + points_read_total);
246
247 points_read_total += points_in_block;
248 points_since_commit += points_in_block;
249
250 if (points_since_commit >= commit_every_n_points &&
251 points_read_total < points_to_read) {
252 commit_and_reopen();
253 start_transaction();
254 points_since_commit = 0;
255 }
256 }
257
258 if (points_read_total != points_to_read)
259 throw std::runtime_error(
260 "Не удалось прочитать весь запрошенный диапазон точек из BIN файла");
261
262 commit_and_reopen();
263 bin_close(reader);
264 } catch (...) {
265 duckdb_disconnect(&conn);
266 duckdb_close(&db);
267 bin_close(reader);
268 throw;
269 }
270
271 duckdb_disconnect(&conn);
272 duckdb_close(&db);
273 }

329 static void duckdb_append_signal_header(duckdb_connection conn, const SignalHeader &header) {
330 duckdb_appender signals_appender = nullptr;
331 if (duckdb_appender_create(conn, nullptr, "signals", &signals_appender) == DuckDBError) {
332 throw std::runtime_error("Не удалось создать DuckDB appender для signals таблицы");
333 }
334
335 std::string optional_json_data = serialize_to_json(header.optional_fields);
336
337 duckdb_appender_begin_row(signals_appender);
338 duckdb_append_double(signals_appender, header.frequency);
339 duckdb_append_double(signals_appender, header.lsb_default);
340 duckdb_append_int64(signals_appender, header.ifirst_point);
341 duckdb_append_varchar(signals_appender, header.time_start.c_str());
342 duckdb_append_varchar(signals_appender, "BINHDR");
343 duckdb_append_varchar(signals_appender, generate_format_version().c_str());
344 duckdb_append_varchar(signals_appender, optional_json_data.c_str());
345 duckdb_appender_end_row(signals_appender);
346
347 if (duckdb_appender_flush(signals_appender) == DuckDBError) {
348 duckdb_error_data error = duckdb_appender_error_data(signals_appender);
349 std::string error_message = duckdb_error_data_message(error);
350 duckdb_destroy_error_data(&error);
351 duckdb_appender_destroy(&signals_appender);
352 throw std::runtime_error(
353 std::format("Не удалось записать метаданные сигнала в DuckDB: {}", error_message));
354 }

358 static void duckdb_append_channels_data(duckdb_connection conn, const SignalHeader &header) {
359 duckdb_appender channels_appender = nullptr;
360 if (duckdb_appender_create(conn, nullptr, "channels", &channels_appender) == DuckDBError) {
361 throw std::runtime_error("Не удалось создать DuckDB appender для channels таблицы");
362 }
363
364 for (int ch = 0; ch < header.num_channels; ch++) {
365 duckdb_appender_begin_row(channels_appender);

370 header.units.empty() ? "" : header.units[ch].c_str());

372 }
373
374 if (duckdb_appender_flush(channels_appender) == DuckDBError) {
375 duckdb_error_data error = duckdb_appender_error_data(channels_appender);
376 std::string error_message = duckdb_error_data_message(error);
377 duckdb_destroy_error_data(&error);
378 duckdb_appender_destroy(&channels_appender);
379 throw std::runtime_error(
380 std::format("Не удалось записать метаданные каналов в DuckDB: {}", error_message));
381 }
382
383 duckdb_appender_destroy(&channels_appender);

402 static void process_column(duckdb_vector vec, int32_t *target, size_t chunk_size) {
403 void *data_ptr = duckdb_vector_get_data(vec);
404 uint64_t *validity = duckdb_vector_get_validity(vec);
405 const auto *source = static_cast<int32_t *>(data_ptr);
406
407 if (validity == nullptr) {
408 std::copy(source, source + chunk_size, target);
409 return;
410 }
411
412 for (size_t row = 0; row < chunk_size; row++) {
413 target[row] = duckdb_validity_row_is_valid(validity, row) ? source[row] : 0;
414 }
415 }

417 static void transfer_chunk_to_channel_data(duckdb_data_chunk &chunk, duckdb_result &res,
418 signals::SignalData &data,
419 duckdb_prepared_statement &stmt, int64_t &offset,
420 const int64_t points_to_read) {
421 while ((chunk = duckdb_stream_fetch_chunk(res)) != nullptr) {
422 size_t chunk_size = duckdb_data_chunk_get_size(chunk);
423 size_t num_columns = duckdb_data_chunk_get_column_count(chunk);
424
425 if (const auto expected_columns = static_cast<decltype(num_columns)>(data.num_channels);
426 num_columns != expected_columns) {
427 duckdb_destroy_data_chunk(&chunk);
428 duckdb_destroy_result(&res);
429 duckdb_destroy_prepare(&stmt);
430 throw std::runtime_error("Количество столбцов в данных не соответствует количеству "
431 "каналов в метаданных");
432 }
433
434 if (offset + static_cast<int64_t>(chunk_size) > points_to_read) {
435 duckdb_destroy_data_chunk(&chunk);
436 duckdb_destroy_result(&res);
437 duckdb_destroy_prepare(&stmt);
438 throw std::runtime_error(
439 "Получен блок данных, выходящий за пределы запрошенного диапазона точек");
440 }
441
442 for (size_t col = 0; col < num_columns; col++) {
443 duckdb_vector vec = duckdb_data_chunk_get_vector(chunk, col);
444
445 int32_t *target = data.raw_channel(static_cast<int>(col)) + offset;
446 process_column(vec, target, chunk_size);
447 }
448
449 offset += static_cast<int64_t>(chunk_size);
450 duckdb_destroy_data_chunk(&chunk);
451 }
452 }

454 static void append_points_block(duckdb_appender points_appender,
455 const std::vector<int32_t> &chunk_buffer,
456 int64_t points_in_block, int32_t num_channels,
457 int64_t first_point_id_in_block) {
458 const auto channels = static_cast<size_t>(num_channels);
459 for (int64_t point = 0; point < points_in_block; point++) {
460 duckdb_appender_begin_row(points_appender);
461 duckdb_append_int64(points_appender, first_point_id_in_block + point);
462
463 const size_t base = static_cast<size_t>(point) * channels;
464 for (size_t ch = 0; ch < channels; ch++) {
465 duckdb_append_int32(points_appender, chunk_buffer[base + ch]);
466 }
467
468 duckdb_appender_end_row(points_appender);
469 }

472 static void duckdb_read_signal_metadata(duckdb_connection conn, SignalHeader &metadata) {
473 metadata = SignalHeader{};
474 duckdb_result res;
475
476 std::string sql_query = "SELECT frequency, lsb_default, \"offset\", time_start, "
477 "optional_params FROM signals LIMIT 1;";
478 if (duckdb_query(conn, sql_query.c_str(), &res) != DuckDBSuccess) {
479 std::string error = duckdb_result_error(&res);
480 duckdb_destroy_result(&res);
481 throw std::runtime_error(std::format("Не удалось прочитать метаданные: {}", error));
482 }
483 if (duckdb_row_count(&res) == 0) {
484 duckdb_destroy_result(&res);
485 throw std::runtime_error("Метаданные не найдены в базе данных");
486 }
487 metadata.frequency = duckdb_value_double(&res, 0, 0);
488 metadata.lsb_default = duckdb_value_double(&res, 1, 0);
489 metadata.ifirst_point = duckdb_value_int64(&res, 2, 0);
490 char *time_start = duckdb_value_varchar(&res, 3, 0);
491 metadata.time_start = time_start ? time_start : "";
492 duckdb_free(static_cast<void *>(time_start));
493 char *optional_json = duckdb_value_varchar(&res, 4, 0);
494 if (optional_json) {
495 metadata.optional_fields = parse_json(optional_json);
496 duckdb_free(static_cast<void *>(optional_json));
497 }
498 duckdb_destroy_result(&res);
499
500 sql_query = "SELECT COUNT(*) FROM channels;";
501 if (duckdb_query(conn, sql_query.c_str(), &res) != DuckDBSuccess) {
502 std::string error = duckdb_result_error(&res);
503 duckdb_destroy_result(&res);
504 throw std::runtime_error(
505 std::format("Не удалось получить количество каналов: {}", error));
506 }
507
508 metadata.num_channels = static_cast<int32_t>(duckdb_value_int64(&res, 0, 0));
509 duckdb_destroy_result(&res);
510
511 if (metadata.num_channels == 0) {
512 throw std::runtime_error("Количество каналов в базе данных равно нулю");
513 }
514
515 sql_query = "SELECT COUNT(*) FROM points;";
516 if (duckdb_query(conn, sql_query.c_str(), &res) != DuckDBSuccess) {
517 std::string error = duckdb_result_error(&res);
518 duckdb_destroy_result(&res);
519 throw std::runtime_error(
520 std::format("Не удалось получить количество точек: {}", error));
521 }
522
523 int64_t num_points = duckdb_value_int64(&res, 0, 0);
524 duckdb_destroy_result(&res);
525
526 if (num_points <= 0) {
527 throw std::runtime_error("Точки не найдены в базе данных");
528 }
529
530 metadata.ilast_point = metadata.ifirst_point + num_points - 1;
531
532 sql_query = "SELECT name, lsb, unit FROM channels ORDER BY rowid;";
533 if (duckdb_query(conn, sql_query.c_str(), &res) != DuckDBSuccess) {
534 std::string error = duckdb_result_error(&res);
535 duckdb_destroy_result(&res);
536 throw std::runtime_error(
537 std::format("Не удалось прочитать метаданные канала: {}", error));
538 }
539 size_t row_count = duckdb_row_count(&res);
540 metadata.channel_names.reserve(row_count);
541 metadata.lsbs.reserve(row_count);
542 metadata.units.reserve(row_count);
543 for (size_t row = 0; row < row_count; row++) {
544 char *name = duckdb_value_varchar(&res, 0, row);
545 double lsb = duckdb_value_double(&res, 1, row);
546 char *unit = duckdb_value_varchar(&res, 2, row);
547
548 metadata.channel_names.emplace_back(name ? name : "");
549 metadata.lsbs.push_back(lsb);
550 metadata.units.emplace_back(unit ? unit : "");
551
552 duckdb_free(static_cast<void *>(name));
553 duckdb_free(static_cast<void *>(unit));
554 }
555 duckdb_destroy_result(&res);
556 }

558 static void open_and_set_config(duckdb_config &config, const std::filesystem::path &duckdb_path,
559 duckdb_database &db, signals::BinReader &reader,
560 duckdb_connection &conn) {
561 if (duckdb_create_config(&config) == DuckDBSuccess) {
562 duckdb_set_config(config, "max_memory", "256MB");
563 if (duckdb_open_ext(duckdb_path.string().c_str(), &db, config, nullptr) !=
564 DuckDBSuccess) {
