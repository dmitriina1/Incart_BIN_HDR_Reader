27 arrow::Status parquet_open(const std::filesystem::path &path, ParquetReader &reader) {
28 reader = ParquetReader{};
29 reader.row_group_cache.clear();
30 reader.row_group_cache_order.clear();
31 ARROW_ASSIGN_OR_RAISE(reader.file, arrow::io::ReadableFile::Open(path.string()))
32 ARROW_ASSIGN_OR_RAISE(reader.reader,
33 parquet::arrow::OpenFile(reader.file, arrow::default_memory_pool()))
34 reader.reader->set_use_threads(true);
35
36 auto file_metadata = reader.reader->parquet_reader()->metadata();
37 if (!file_metadata)
38 return arrow::Status::Invalid("Не удалось получить метаданные Parquet файла");
39
40 reader.num_row_groups = reader.reader->num_row_groups();
41 reader.row_group_first_row.reserve(static_cast<size_t>(reader.num_row_groups));
42 reader.row_group_row_count.reserve(static_cast<size_t>(reader.num_row_groups));
43
44 int64_t row_offset = 0;
45 for (int rg = 0; rg < reader.num_row_groups; rg++) {
46 const int64_t rg_rows = file_metadata->RowGroup(rg)->num_rows();
47 reader.row_group_first_row.push_back(row_offset);
48 reader.row_group_row_count.push_back(rg_rows);
49 row_offset += rg_rows;
50 }
51
52 return arrow::Status::OK();

65 arrow::Status parquet_write_signals_data(const std::filesystem::path &bin_path,
66 const std::filesystem::path &parquet_path,
67 int64_t point_off, int64_t point_cnt,
68 int64_t row_group_points) {
69 if (point_off < 0)
70 return arrow::Status::Invalid("Смещение точек не может быть отрицательным");
71
72 const int64_t real_row_group_points =
73 row_group_points > 0 ? row_group_points : DEFAULT_ROW_GROUP_POINTS;
74
75 BinReader reader;
76 try {
77 bin_open(bin_path, reader);
78 } catch (const std::exception &e) {
79 return arrow::Status::IOError(std::format("Не удалось открыть BIN файл: {}", e.what()));
80 }
81
82 SignalHeader range_header;
83 const int64_t points_to_write = make_header_for_range(reader.header, reader.total_points,
84 point_off, point_cnt, range_header);
85
86 const int32_t num_chan = range_header.num_channels;
87 const std::shared_ptr<arrow::Schema> schema = make_signal_schema(range_header);
88
89 parquet::WriterProperties::Builder builder;
90 builder.compression(parquet::Compression::ZSTD);
91 builder.encoding(parquet::Encoding::DELTA_BINARY_PACKED);
92 builder.max_row_group_length(real_row_group_points);
93
94 parquet::ArrowWriterProperties::Builder arrow_builder;
95 arrow_builder.store_schema();
96
97 ARROW_ASSIGN_OR_RAISE(auto output_file,
98 arrow::io::FileOutputStream::Open(parquet_path.string()))
99 ARROW_ASSIGN_OR_RAISE(auto writer, parquet::arrow::FileWriter::Open(
100 *schema, arrow::default_memory_pool(), output_file,
101 builder.build(), arrow_builder.build()))
102 ARROW_RETURN_NOT_OK(writer->AddKeyValueMetadata(make_file_metadata(range_header)));
103
104 std::vector<std::vector<int32_t>> buffers(
105 static_cast<size_t>(num_chan),
106 std::vector<int32_t>(static_cast<size_t>(real_row_group_points)));
107 std::vector<int32_t> raw_chunk;
108 raw_chunk.reserve(static_cast<size_t>(real_row_group_points) *
109 static_cast<size_t>(num_chan));
110
111 bin_seek(reader, point_off);
112 int64_t points_written = 0;
113
114 while (points_written < points_to_write) {
115 const int64_t chunk_size =
116 std::min(real_row_group_points, points_to_write - points_written);
117 const int64_t points_read = bin_read_data_chunk(reader, chunk_size, raw_chunk);
118 if (points_read <= 0)
119 break;
120
121 for (int64_t p = 0; p < points_read; p++) {
122 const size_t src = static_cast<size_t>(p) * static_cast<size_t>(num_chan);
123 for (int ch = 0; ch < num_chan; ch++)
124 buffers[static_cast<size_t>(ch)][static_cast<size_t>(p)] =
125 raw_chunk[src + static_cast<size_t>(ch)];
126 }
127
128 std::vector<std::shared_ptr<arrow::Array>> arrays;
129 arrays.reserve(static_cast<size_t>(num_chan));
130 for (int ch = 0; ch < num_chan; ch++) {
131 arrow::Int32Builder intBuilder;
132 ARROW_RETURN_NOT_OK(
133 intBuilder.AppendValues(buffers[static_cast<size_t>(ch)].data(), points_read));
134 std::shared_ptr<arrow::Array> arr;
135 ARROW_RETURN_NOT_OK(intBuilder.Finish(&arr));
136 arrays.push_back(arr);
137 }
138
139 auto batch = arrow::RecordBatch::Make(schema, points_read, arrays);
140 ARROW_RETURN_NOT_OK(writer->WriteRecordBatch(*batch));
141 points_written += points_read;
142 }
143
144 if (points_written != points_to_write) {
145 bin_close(reader);
146 return arrow::Status::IOError(
147 "Не удалось записать весь запрошенный диапазон точек в Parquet");
148 }
149
150 bin_close(reader);
151 return writer->Close();
152 }

166 arrow::Status parquet_read_signal_data_like_stream(ParquetReader &reader, SignalData &data,
167 int64_t point_off, int64_t point_cnt) {
168 if (point_off < 0)
169 return arrow::Status::Invalid("Смещение не может быть отрицательным");
170
171 if (reader.num_row_groups == 0)
172 return arrow::Status::Invalid("Parquet файл не содержит row groups");
173
174 ARROW_ASSIGN_OR_RAISE(auto metadata, read_parquet_metadata(reader))
175
176 const int64_t total_rows = reader.reader->parquet_reader()->metadata()->num_rows();
177 const int64_t points_to_read =
178 prepare_signal_data_window(metadata, total_rows, point_off, point_cnt, data, true);
179 if (points_to_read == 0)
180 return arrow::Status::OK();
181
182 if (reader.row_group_first_row.size() != static_cast<size_t>(reader.num_row_groups) ||
183 reader.row_group_row_count.size() != static_cast<size_t>(reader.num_row_groups)) {
184 return arrow::Status::Invalid("Parquet row-group метаданные некорректные");
185 }
186
187 const int64_t target_end = point_off + points_to_read;
188
189 const auto first_it = std::ranges::upper_bound(reader.row_group_first_row, point_off);
190 auto first_rg =
191 static_cast<int>(std::distance(reader.row_group_first_row.begin(), first_it));
192 if (first_rg > 0)
193 first_rg--;
194
195 for (int rg = first_rg; rg < reader.num_row_groups; rg++) {
196 const int64_t rg_start = reader.row_group_first_row[static_cast<size_t>(rg)];
197 if (rg_start >= target_end)
198 break;
199
200 const int64_t rg_rows = reader.row_group_row_count[static_cast<size_t>(rg)];
201
202 std::shared_ptr<arrow::Table> table = read_row_group(reader, rg);
203 if (!table)
204 return arrow::Status::Invalid(std::format("Не удалось прочитать row group {}", rg));
205
206 const int64_t local_start = std::max<int64_t>(0, point_off - rg_start);
207 const int64_t local_end = std::min<int64_t>(rg_rows, target_end - rg_start);
208 const int64_t copy_count = local_end - local_start;
209
210 const int64_t out_off = rg_start + local_start - point_off;
211
212 for (int ch = 0; ch < data.num_channels; ch++) {
213 ARROW_RETURN_NOT_OK(
214 copy_column_from_table(table, ch, local_start, copy_count, out_off, data));
215 }
216 }
217
218 return arrow::Status::OK();
219 }

244 int row_group_index) {
245 if (auto it = reader.row_group_cache.find(row_group_index);
246 it != reader.row_group_cache.end()) {
247 reader.row_group_cache_order.erase(
248 std::ranges::find(reader.row_group_cache_order, row_group_index));
249 reader.row_group_cache_order.push_back(row_group_index);

253 // ReadRowGroup - deprecated, так что читаем через RecordBatchReader, который должен
254 // использовать оптимизированные методы чтения row group std::shared_ptr<arrow::Table>
255 // table; if (auto st = reader.reader->ReadRowGroup(row_group_index, &table); !st.ok() ||
256 // !table)
257 // return nullptr;
258
259 std::vector<int> row_groups{ row_group_index };
260
261 auto batch_reader_result = reader.reader->GetRecordBatchReader(row_groups);
262 if (!batch_reader_result.ok())
263 return nullptr;
264
265 std::unique_ptr<arrow::RecordBatchReader> batch_reader =
266 batch_reader_result.MoveValueUnsafe();
267
268 if (!batch_reader)
269 return nullptr;
270
271 auto table_result = arrow::Table::FromRecordBatchReader(batch_reader.get());
272 if (!table_result.ok())
273 return nullptr;
274
275 std::shared_ptr<arrow::Table> table = table_result.MoveValueUnsafe();
276 if (!table)
277 return nullptr;
278
279 if (reader.row_group_cache.size() >= MAX_ROW_GROUP_CACHE) {
280 int lru = reader.row_group_cache_order.front();
281 reader.row_group_cache.erase(lru);
282 reader.row_group_cache_order.erase(reader.row_group_cache_order.begin());
283 }
284
285 reader.row_group_cache[row_group_index] = table;
286 reader.row_group_cache_order.push_back(row_group_index);
287 return table;

317 static std::shared_ptr<arrow::Schema> make_signal_schema(const SignalHeader &header) {
318 const int32_t num_channels = header.num_channels;
319 std::vector<std::shared_ptr<arrow::Field>> fields;
320 fields.reserve(static_cast<size_t>(num_channels));

322 for (int32_t ch = 0; ch < num_channels; ch++) {

336 field_metadata->Append("type", "int32");
337 field_metadata->Append("unit", unit);
338
339 fields.push_back(arrow::field(channel_name, arrow::int32(), false, field_metadata));
340 }
341
342 return arrow::schema(fields);

345 static arrow::Result<SignalHeader> read_parquet_metadata(ParquetReader &reader) {
346 auto file_metadata = reader.reader->parquet_reader()->metadata();
347 if (!file_metadata) {
348 return arrow::Status::Invalid("Не удалось получить метаданные Parquet файла");
349 }
350 auto map = file_metadata->key_value_metadata();
351 if (!map) {
352 return arrow::Status::Invalid("Parquet файл не содержит ключ-значение метаданных");
353 }
354
355 auto get_field = [&](const std::string &key) {
356 auto result = map->Get(key);
357 return result.ok() ? *result : "";
358 };
359
360 SignalHeader metadata;
361 metadata.num_channels = static_cast<int32_t>(file_metadata->schema()->num_columns());
362 metadata.frequency = std::stod(get_field("frequency"));
363 metadata.lsb_default = std::stod(get_field("lsb_default"));
364 metadata.ifirst_point = std::stoll(get_field("offset"));
365 metadata.ilast_point = metadata.ifirst_point + file_metadata->num_rows() - 1;
366 metadata.time_start = get_field("time_start");
367
368 std::shared_ptr<arrow::Schema> schema;
369 ARROW_RETURN_NOT_OK(reader.reader->GetSchema(&schema));
370
371 metadata.channel_names.reserve(metadata.num_channels);
372 metadata.lsbs.reserve(metadata.num_channels);
373 metadata.units.reserve(metadata.num_channels);
374
375 for (int32_t ch = 0; ch < metadata.num_channels; ch++) {
376 auto field = schema->field(ch);
377 auto field_map = field->metadata();
378 metadata.channel_names.push_back(field->name());
379 metadata.lsbs.push_back(std::stod(field_map->Get("lsb").ValueOrDie()));
380 metadata.units.push_back(field_map->Get("unit").ValueOrDie());
381 }
382 return metadata;
383 }

385 static arrow::Status copy_column_from_table(const std::shared_ptr<arrow::Table> &table,
386 int channel, int64_t local_start,
387 int64_t copy_count, int64_t out_offset,
388 SignalData &data) {
389 int32_t *dst = data.raw_channel(channel) + out_offset;
390 auto chunked = table->column(channel);
391 int64_t skip = local_start;
392 int64_t written = 0;
393
394 for (const auto &chunk : chunked->chunks()) {
395 const int64_t len = chunk ? chunk->length() : 0;
396 if (skip >= len) {
397 skip -= len;
398 continue;
399 }
400 const int64_t take = std::min(copy_count - written, len - skip);
401 auto int_arr = std::dynamic_pointer_cast<arrow::Int32Array>(chunk);
402
403 const int32_t *src = int_arr->raw_values() + skip;
404 for (int64_t p = 0; p < take; p++) {
405 dst[written + p] = src[p];
406 }
407 written += take;
408 skip = 0;
409 if (written == copy_count)
410 break;
411 }
412 return arrow::Status::OK();
413 }
