7 int64_t normalize_range(int64_t total_points, int64_t point_off, int64_t point_cnt) {
8 const int64_t safe_total_points = std::max<int64_t>(0, total_points);
9 if (point_off >= safe_total_points) {
10 return 0;
11 }

13 const int64_t available = safe_total_points - point_off;
14 return (point_cnt < 0) ? available : std::min(point_cnt, available);
