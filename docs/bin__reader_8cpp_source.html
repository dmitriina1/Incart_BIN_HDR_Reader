51 void bin_seek(BinReader &reader, int64_t point_off) {
52 if (!reader.file.is_open()) {
53 throw std::logic_error("BIN reader не открыт");
54 }
55 if (point_off < 0) {
56 throw std::invalid_argument("Смещение точек должно быть неотрицательным");
57 }
58 if (point_off > reader.total_points) {
59 point_off = reader.total_points;
60 }
61
62 const int64_t frame_size = static_cast<int64_t>(reader.header.num_channels) *
63 static_cast<int64_t>(sizeof(int32_t));
64 const int64_t bytes_offset = point_off * frame_size;
65
66 reader.file.clear();
67 reader.file.seekg(static_cast<std::streamoff>(bytes_offset));
68 if (!reader.file) {
69 throw std::runtime_error(
70 std::format("Не удалось установить позицию чтения в BIN файле (смещение {} байт)",
71 bytes_offset));
72 }
73
74 reader.current_point = point_off;

77 int64_t bin_read_data_chunk(BinReader &reader, int64_t point_cnt,
78 std::vector<int32_t> &block_buf) {
79 if (!reader.file.is_open()) {
80 throw std::logic_error("BIN reader не открыт");
81 }
82 if (point_cnt == 0) {
83 block_buf.clear();
84 return 0;
85 }
86
87 const int64_t points_to_read =
88 normalize_range(reader.total_points, reader.current_point, point_cnt);
89 if (points_to_read == 0) {
90 block_buf.clear();
91 return 0;
92 }
93
94 block_buf.resize(static_cast<size_t>(points_to_read) *
95 static_cast<size_t>(reader.header.num_channels));

97 reader.current_point += points_to_read;

101 void bin_read(const std::filesystem::path &bin_path, int64_t point_off, int64_t point_cnt,
102 SignalData &data) {
103 BinReader reader;
104 bin_open(bin_path, reader);
105
106 const int64_t points_to_read = normalize_range(reader.total_points, point_off, point_cnt);
107 bin_seek(reader, point_off);
108
109 data = SignalData{};
110 static_cast<SignalHeader &>(data) = reader.header;
111 data.ifirst_point = reader.header.ifirst_point + point_off;
112 data.ilast_point = (points_to_read > 0) ? (data.ifirst_point + points_to_read - 1)
113 : (data.ifirst_point - 1);
114 data.num_points = points_to_read;
115
116 if (points_to_read > 0) {
117 allocate_storage(data, points_to_read);
118
119 std::vector<int32_t> block_buf;
120 int64_t point_pos = 0;
121 while (point_pos < points_to_read) {
122 const int64_t block_points =
123 std::min<int64_t>(BIN_READ_CHUNK_SIZE, points_to_read - point_pos);
124 const int64_t points_in_block =
125 bin_read_data_chunk(reader, block_points, block_buf);
126 if (points_in_block == 0) {
127 break;
128 }
129 copy_raw_data(block_buf, data, point_pos, points_in_block);
130 point_pos += points_in_block;
131 }
132 }
133 bin_close(reader);
134 }

155 int64_t points_to_read, int64_t num_channels) {
156 const auto bytes_to_read =
157 static_cast<std::streamsize>(points_to_read * num_channels * sizeof(int32_t));

159 file.read(reinterpret_cast<char *>(block_buf.data()), bytes_to_read); // NOSONAR

168 static void copy_raw_data(const std::vector<int32_t> &block_buf, SignalData &data,
169 int64_t dest_offset, int64_t points_in_block) {
170 if (data.lsbs.empty())
171 data.lsbs.assign(static_cast<size_t>(data.num_channels), data.lsb_default);
172
173 const int64_t num_channels = data.num_channels;
174
175 for (int ch = 0; ch < num_channels; ch++) {
176 int32_t *destptr = data.raw_channel(ch) + dest_offset;
177 for (int64_t s = 0; s < points_in_block; s++)
178 destptr[s] =
179 block_buf[(static_cast<size_t>(s) * static_cast<size_t>(num_channels)) +
180 static_cast<size_t>(ch)];
181 }
