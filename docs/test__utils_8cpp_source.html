175 bool compare_signal_data(const SignalData &expected, const SignalData &actual,
176 const char *label) {
177 if (expected.num_points != actual.num_points ||
178 expected.num_channels != actual.num_channels) {
179 std::fprintf(stderr, "Ошибка (%s): точек или каналов не совпадает: %lld vs %lld\n",
180 label, static_cast<long long>(expected.num_points),
181 static_cast<long long>(actual.num_points));
182 return false;
183 }
184
185 if (expected.ifirst_point != actual.ifirst_point ||
186 expected.ilast_point != actual.ilast_point) {
187 std::fprintf(stderr, "Ошибка (%s): диапазоны не совпадают: %lld..%lld vs %lld..%lld\n",
188 label, static_cast<long long>(expected.ifirst_point),
189 static_cast<long long>(expected.ilast_point),
190 static_cast<long long>(actual.ifirst_point),
191 static_cast<long long>(actual.ilast_point));
192 return false;
193 }
194
195 for (int ch = 0; ch < expected.num_channels; ch++) {
196 for (int64_t p = 0; p < expected.num_points; p++) {
197 if (expected.raw_channel(ch)[p] != actual.raw_channel(ch)[p]) {
198 std::fprintf(stderr,
199 "Ошибка (%s): несовпадение на канале %d, точка %lld: %f vs %f\n",
200 label, ch, static_cast<long long>(p), expected.value(ch, p),
201 actual.value(ch, p));
202 return false;
203 }
204 }
205 }
206 return true;
207 }

425 int64_t chunk_offset, int64_t chunk_count, int64_t chunk_index,

444 "%lld: ожидалось %d, а сейчас %d\n",
445 chunk_index, order_index, chunk_offset, ch, chunk_offset + point,
446 reference.raw_channel(ch)[chunk_offset + point],
447 chunk_data.raw_channel(ch)[point]);
448 return false;
449 }

455 static double benchmark_chunk_reading_impl(
456 const std::function<void(const std::filesystem::path &, SignalData &, int64_t, int64_t)>
457 &read_signal,
458 const std::filesystem::path &source_path, const SignalData &reference, int64_t chunk_points,
459 bool random_order) {
460 if (reference.num_points <= 0 || reference.num_channels <= 0)
461 return 0.0;
462
463 const int64_t safe_chunk_points =
464 normalize_chunk_points(chunk_points, reference.num_points);
465 const int64_t num_chunks =
466 (reference.num_points + safe_chunk_points - 1) / safe_chunk_points;
467
468 std::vector<int64_t> chunk_order;
469 if (random_order)
470 chunk_order = make_chunk_order(num_chunks, true);
471
472 SignalData chunk_data;
473 chunk_data.storage.reserve(static_cast<size_t>(reference.num_channels) *
474 static_cast<size_t>(safe_chunk_points));
475
476 const auto t0 = std::chrono::steady_clock::now();
477 for (int64_t i = 0; i < num_chunks; i++) {
478 const int64_t chunk_index = random_order ? chunk_order[static_cast<size_t>(i)] : i;
479 const int64_t chunk_offset = chunk_index * safe_chunk_points;
480 const int64_t chunk_count =
481 std::min(safe_chunk_points, reference.num_points - chunk_offset);
482 read_signal(source_path, chunk_data, chunk_offset, chunk_count);
483 }
484
485 return std::chrono::duration<double>(std::chrono::steady_clock::now() - t0).count();
486 }
